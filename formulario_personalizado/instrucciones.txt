function doPost(e) {

    var hoja = SpreadsheetApp
        .getActiveSpreadsheet()
        .getSheetByName("Ventas");

    var modelo = e.parameter.modelo;
    var precio = e.parameter.precio;
    var cliente = e.parameter.cliente;

    hoja.appendRow([
        modelo,
        precio,
        cliente
    ]);

    return ContentService.createTextOutput("Venta guardada");
}

var formulario = document.getElementById("formulario");

formulario.addEventListener("submit", function(event) {

    event.preventDefault();

    var modelo = document.getElementById("modelo").value;
    var precio = document.getElementById("precio").value;
    var cliente = document.getElementById("cliente").value;

    var datos = new URLSearchParams();

    datos.append("modelo", modelo);
    datos.append("precio", precio);
    datos.append("cliente", cliente);

    fetch("PEGA_AQUI_TU_URL", {
        method: "POST",
        mode: "no-cors",
        body: datos
    });

    alert("Venta guardada");

});
